// ═══════════════════════════════════════════════════════════════════════════ // xero-proxy — Supabase Edge Function for ShiftOps // Creates invoices in Xero directly, so "Create & Send via Xero" fills the // contact, dates, reference and EVERY line item automatically instead of // opening a blank Xero page. // // Deploy: supabase functions deploy xero-proxy --no-verify-jwt // Secrets: supabase secrets set XERO_CLIENT_ID=... XERO_CLIENT_SECRET=... // Connect: open the function URL once in a browser and approve in Xero: // https://jpfnrsxnnbtcadsnjrpc.supabase.co/functions/v1/xero-proxy // // Tokens are stored in the existing app_settings table (key='xero_oauth') via // the service-role key — no schema change. Xero ROTATES refresh tokens on // every use, so the new one is persisted after every refresh. // ═══════════════════════════════════════════════════════════════════════════ import { createClient } from "npm:@supabase/supabase-js@2"; const XERO_CLIENT_ID = Deno.env.get("XERO_CLIENT_ID") ?? ""; const XERO_CLIENT_SECRET = Deno.env.get("XERO_CLIENT_SECRET") ?? ""; const SUPABASE_URL = Deno.env.get("SUPABASE_URL")!; const SERVICE_ROLE = Deno.env.get("SUPABASE_SERVICE_ROLE_KEY")!; // Apps created after 2 March 2026 only support Xero's new GRANULAR scopes — // the broad accounting.transactions is rejected with invalid_scope. Creating // invoices needs accounting.invoices; contacts for name→contact matching. const SCOPES = "offline_access accounting.invoices accounting.contacts"; const db = createClient(SUPABASE_URL, SERVICE_ROLE); const CORS = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Headers": "authorization, x-client-info, apikey, content-type", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", }; function json(body: unknown, status = 200) { return new Response(JSON.stringify(body), { status, headers: { ...CORS, "Content-Type": "application/json" }, }); } function html(body: string, status = 200) { return new Response( `
${body}
`, { status, headers: { ...CORS, "Content-Type": "text/html; charset=utf-8" } }, ); } // ── Token storage (app_settings key='xero_oauth') ─────────────────────────── async function getStoredTokens(): Promise<{ refresh_token: string; tenant_id: string } | null> { const { data } = await db.from("app_settings").select("value").eq("key", "xero_oauth").maybeSingle(); const v = data?.value; return v?.refresh_token ? v : null; } async function storeTokens(v: Record) { await db.from("app_settings").upsert( { key: "xero_oauth", value: v, updated_at: new Date().toISOString() }, { onConflict: "key" }, ); } // ── OAuth helpers ──────────────────────────────────────────────────────────── function basicAuth() { return "Basic " + btoa(`${XERO_CLIENT_ID}:${XERO_CLIENT_SECRET}`); } async function tokenRequest(params: Record) { const resp = await fetch("https://identity.xero.com/connect/token", { method: "POST", headers: { Authorization: basicAuth(), "Content-Type": "application/x-www-form-urlencoded" }, body: new URLSearchParams(params), }); const data = await resp.json(); if (!resp.ok) throw new Error(`Xero token error: ${data.error || resp.status}`); return data; } // Returns a fresh access token + tenant id. Persists the ROTATED refresh token. async function getAccessToken(): Promise<{ access_token: string; tenant_id: string }> { const stored = await getStoredTokens(); if (!stored) throw new Error("Xero not connected yet — open the xero-proxy function URL in a browser and approve access."); const tok = await tokenRequest({ grant_type: "refresh_token", refresh_token: stored.refresh_token }); // Xero rotates refresh tokens — losing the new one breaks the connection. await storeTokens({ ...stored, refresh_token: tok.refresh_token, updated_at: new Date().toISOString() }); return { access_token: tok.access_token, tenant_id: stored.tenant_id }; } // ═══════════════════════════════════════════════════════════════════════════ Deno.serve(async (req) => { if (req.method === "OPTIONS") return new Response("ok", { headers: CORS }); const url = new URL(req.url); // The Edge runtime strips the /functions/v1 prefix from req.url, so building // redirect_uri from url.pathname produced .../xero-proxy WITHOUT the prefix — // which Xero rejects as "Invalid redirect_uri". Build it from the project URL // instead so it always matches the URI registered in the Xero app. const selfUrl = `${SUPABASE_URL}/functions/v1/xero-proxy`; if (!XERO_CLIENT_ID || !XERO_CLIENT_SECRET) { return json({ error: "XERO_CLIENT_ID / XERO_CLIENT_SECRET secrets not set" }, 500); } // ── GET: one-time browser consent flow ──────────────────────────────────── if (req.method === "GET") { const code = url.searchParams.get("code"); if (!code) { // Already connected? Show status instead of re-prompting. const stored = await getStoredTokens(); if (stored && !url.searchParams.get("reconnect")) { return html(`

✅ Xero is connected

ShiftOps can create draft invoices in Xero.

To reconnect a different org, add ?reconnect=1 to this URL.

`); } // Build the consent URL manually: URLSearchParams encodes the spaces // between scope names as '+', which Xero's login server rejects with // "invalid_scope". encodeURIComponent produces %20, which it accepts. // ?scope=... overrides the default scopes — for diagnosing which scopes // this particular Xero app registration is allowed to request. const scopeToUse = url.searchParams.get("scope") || SCOPES; const consent = "https://login.xero.com/identity/connect/authorize" + `?response_type=code` + `&client_id=${encodeURIComponent(XERO_CLIENT_ID)}` + `&redirect_uri=${encodeURIComponent(selfUrl)}` + `&scope=${encodeURIComponent(scopeToUse)}` + `&state=${crypto.randomUUID()}`; // ?debug=1 → show exactly what would be sent to Xero instead of redirecting. if (url.searchParams.get("debug")) { return json({ deployed_version: "v3-debug", client_id_prefix: XERO_CLIENT_ID.slice(0, 8) + "…", redirect_uri: selfUrl, scope: SCOPES, consent_url: consent, }); } return Response.redirect(consent, 302); } // Callback: exchange code, discover tenant, store refresh token. try { const tok = await tokenRequest({ grant_type: "authorization_code", code, redirect_uri: selfUrl }); const connResp = await fetch("https://api.xero.com/connections", { headers: { Authorization: `Bearer ${tok.access_token}` }, }); const conns = await connResp.json(); const tenant = Array.isArray(conns) && conns[0]; if (!tenant?.tenantId) throw new Error("No Xero organisation authorised"); await storeTokens({ refresh_token: tok.refresh_token, tenant_id: tenant.tenantId, tenant_name: tenant.tenantName || "", connected_at: new Date().toISOString(), }); // Redirect to the clean status URL so the one-time code is no longer in // the address bar — reloading the success page can't re-fire the exchange. return Response.redirect(selfUrl, 302); } catch (e) { // Xero auth codes are SINGLE-USE: reloading the callback re-sends the // same code and Xero rejects it with invalid_grant. If tokens are already // stored, the original exchange succeeded — show connected, not an error. const stored = await getStoredTokens(); if (stored) return Response.redirect(selfUrl, 302); return html(`

❌ Connection failed

${(e as Error).message}

Try again

`, 500); } } // ── POST: proxied Xero API call from ShiftOps ───────────────────────────── if (req.method === "POST") { try { const body = await req.json().catch(() => ({})); // Health probe used by the front-end / manual testing. if (body.action === "status") { const stored = await getStoredTokens(); return json({ connected: !!stored, tenant: stored?.tenant_id ? "set" : null }); } const endpoint = String(body.endpoint || ""); if (!/^[A-Za-z]+$/.test(endpoint)) return json({ error: "Invalid endpoint" }, 400); const method = body.method === "POST" ? "POST" : "GET"; const { access_token, tenant_id } = await getAccessToken(); const xeroResp = await fetch(`https://api.xero.com/api.xro/2.0/${endpoint}`, { method, headers: { Authorization: `Bearer ${access_token}`, "Xero-tenant-id": tenant_id, Accept: "application/json", ...(method === "POST" ? { "Content-Type": "application/json" } : {}), }, body: method === "POST" ? JSON.stringify(body.payload ?? {}) : undefined, }); const text = await xeroResp.text(); let data: unknown; try { data = JSON.parse(text); } catch { data = { raw: text }; } if (!xeroResp.ok) { return json({ error: `Xero API ${xeroResp.status}`, detail: data }, xeroResp.status); } return json(data); } catch (e) { return json({ error: (e as Error).message }, 500); } } return json({ error: "Method not allowed" }, 405); });
